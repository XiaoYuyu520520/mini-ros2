     99 def rclpy_argv():
    100     # 这个脚本的航点文件是个裸参数，rclpy 不认识它（真车上 `ros2 run racecar go.py FILE`
    101     # 会把 FILE 当成解析不了的全局参数），所以裸参数不进 rclpy；带 `-` 的照旧传进去。
    102     # 但 `-p use_sim_time:=true` 是「选项 + 它的值」，只留带 `-` 的那半会剩下一个没值的
    103     # `-p`（兼容层的 rclpy 会安静地丢掉它，于是脚本留在墙上时间上）。所以 `--ros-args`
    104     # 往后原样透传，让 -p/-r 的取值跟着走。
    105     argv = sys.argv
    106     tail = []
    107     if '--ros-args' in argv:
    108         cut = argv.index('--ros-args')
    109         tail, argv = argv[cut:], argv[:cut]
    110     return [argv[0]] + [a for a in argv[1:] if a.startswith('-')] + tail
